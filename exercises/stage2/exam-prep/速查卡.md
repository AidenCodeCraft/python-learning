# 二、Redis 5 类结构（命令 → Python）

```python
# String：验证码/计数
r.set("code:手机号", "123456", ex=300)
r.get("code:手机号"); r.incr("course:1:visits")
r.set("lock:k", "1", nx=True, ex=10)      # 不存在才写

# Hash：学员资料
r.hset("student:1", mapping={"name": "张三", "phone": "138"})
r.hget("student:1", "name"); r.hgetall("student:1")   # dict
r.hset("student:1", "phone", "139"); r.hincrby("student:1", "hours", 2)
r.hdel("student:1", "phone"); r.hexists("student:1", "name")

# List：队列/最近 N 条
r.rpush("queue", "a", "b"); r.lpop("queue")            # 先进先出
r.lrange("queue", 0, -1); r.llen("queue"); r.ltrim("recent", 0, 9)

# Set：去重/集合运算
r.sadd("course:1", "s1", "s2"); r.smembers("course:1")
r.sismember("course:1", "s1"); r.scard("course:1")
r.sinter("course:1", "course:2")                       # 交集=同时报名

# Sorted Set：排行榜
r.zadd("rank", {"张三": 85, "李四": 90})
r.zincrby("rank", 5, "张三"); r.zscore("rank", "张三")
r.zrevrange("rank", 0, 2, withscores=True)             # 前三名（降序）
rank0 = r.zrevrank("rank", "张三")                      # 0 开始 → 名次=rank0+1
r.zrem("rank", "张三")

# Pipeline
pipe = r.pipeline(transaction=True); pipe.set("k1", "v1"); pipe.incr("cnt"); pipe.execute()
```

**理解题一句话答案**：
- String vs Hash：整块读写用 String；单独改字段用 Hash
- List 有序可重复；Set 无序不重复
- 排行榜用 ZSet：自带分数排序，取前 N / 查排名快
- 排名从 **0** 开始，**+1** 才是名次
- Pipeline = 减少网络往返；事务 = MULTI/EXEC 按序执行（无回滚）
- 先 GET 再 SET 计数会丢更新 → 用 **INCR**（原子）

## 事务（报名扣款，考点必考）

```python
conn = pymysql.connect(**DB_CONF)          # autocommit=False
try:
    with conn.cursor() as cur:
        cur.execute("SELECT price FROM courses WHERE id = %s", (cid,))
        row = cur.fetchone()
        if row is None:
            conn.rollback(); return False
        cur.execute("UPDATE accounts SET balance = balance - %s WHERE student_id = %s AND balance >= %s",
                    (row["price"], sid, row["price"]))
        if cur.rowcount == 0:
            conn.rollback(); return False
        cur.execute("INSERT INTO enrollments (student_id, course_id, status) VALUES (%s, %s, 1)", (sid, cid))
    conn.commit(); return True
except pymysql.err.IntegrityError:
    conn.rollback(); return False          # 重复报名
except Exception:
    conn.rollback(); raise
finally:
    conn.close()
```

# 四、SQL 语法片段（可能考）

```sql
-- 三表 JOIN
SELECT s.name, c.name, e.score
FROM enrollments e
JOIN students s ON s.id = e.student_id
JOIN courses c  ON c.id = e.course_id;

-- 左连接：所有学员含未报名
FROM students s LEFT JOIN enrollments e ON e.student_id = s.id;

-- 高于本课程平均分（相关子查询）
WHERE e.score > (SELECT AVG(e2.score) FROM enrollments e2 WHERE e2.course_id = e.course_id)

-- 从未报名
WHERE NOT EXISTS (SELECT 1 FROM enrollments e WHERE e.student_id = s.id)

-- 分组过滤
GROUP BY course_id HAVING COUNT(*) > 2

-- CASE
CASE WHEN score >= 85 THEN '优秀' WHEN score >= 60 THEN '及格' ELSE '不及格' END

-- 分页（时间倒序，同时间按 id）
ORDER BY created_at DESC, id DESC LIMIT 0, 5

-- upsert
INSERT INTO enrollments (...) VALUES (...)
ON DUPLICATE KEY UPDATE score = VALUES(score), status = VALUES(status);

-- 多表更新
UPDATE enrollments e JOIN courses c ON c.id = e.course_id
SET e.status = 1 WHERE c.name = 'Python入门' AND e.status = 0;
```

# 六、易错点清单（考前扫一眼）

| 易错 | 正确 |
|---|---|
| `WHERE` 写 `= NULL` | `IS NULL` |
| SQL 用 f-string 拼值 | 参数化 `%s` |
| UPDATE/DELETE 忘 WHERE | 写完回头看 |
| 事务里忘了 `commit()` | 提交后数据才生效 |
| 缓存先删再更新库 | **先更库后删缓存** |
| 中文变 `\u6c64` | `ensure_ascii=False` |
| Redis 返回 `b'...'` | 连接加 `decode_responses=True` |
| 排名直接用 `zrevrank` | **+1** 才是名次 |
| 没判断影响行数 | `cur.rowcount == 0` → 回滚/提示 |
| 连接没关 | `finally: conn.close()` |
